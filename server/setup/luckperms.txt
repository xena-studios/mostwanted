# Ranks. Run each `lp` line in the server console once, after the first start
# with the database connected. Colours follow the in-game palette
# (docs/09-brand.md): inmates orange (hue 30), guards blue (hue 210), staff light
# purple (hue 300). The prefixes are for name tags and the tab list (drawn by core);
# chat tags come from zChat's formats,
# which each rank turns on with zchat.group.<rank>.

# Everyone: inmates and criminals. The core plugin switches between the two; it isn't a rank.
lp group default setweight 0
lp group default meta setprefix 0 "<#ff9933>ɪɴᴍᴀᴛᴇ</#ff9933> "

# Guards: picked by application, synced from the Discord role by the MostWanted bot (backend/).
lp creategroup guard
lp group guard permission set zchat.group.guard true
lp group guard setweight 10
lp group guard parent add default
lp group guard meta setprefix 10 "<#3399ff>ɢᴜᴀʀᴅ</#3399ff> "
lp group guard permission set mostwanted.guard true

# Staff.
lp creategroup helper
lp group helper permission set zchat.group.helper true
lp group helper setweight 50
lp group helper parent add default
lp group helper meta setprefix 50 "<#ff33ff>ʜᴇʟᴘᴇʀ</#ff33ff> "

lp creategroup moderator
lp group moderator permission set zchat.group.moderator true
lp group moderator setweight 60
lp group moderator parent add helper
lp group moderator meta setprefix 60 "<#ff33ff>ᴍᴏᴅ</#ff33ff> "
lp group moderator permission set libertybans.* true
lp group moderator permission set coreprotect.inspect true
lp group moderator permission set coreprotect.lookup true
lp group moderator permission set grim.alerts true

lp creategroup admin
lp group admin permission set zchat.group.admin true
lp group admin setweight 90
lp group admin parent add moderator
lp group admin meta setprefix 90 "<#ff33ff>ᴀᴅᴍɪɴ</#ff33ff> "
lp group admin permission set mostwanted.admin true
lp group admin permission set coreprotect.* true

lp creategroup owner
lp group owner permission set zchat.group.owner true
lp group owner setweight 100
lp group owner parent add admin
lp group owner meta setprefix 100 "<#ff33ff>ᴏᴡɴᴇʀ</#ff33ff> "
lp group owner permission set * true

# Store ranks (website/src/store/catalog.ts, docs/11-store.md). Tebex adds them
# with `lp user {username} parent add <rank>`. Their tag is a suffix, after the
# name, so the role tag in front (inmate, guard, staff) always shows. Each rank
# includes the one before it. Cosmetics are mostwanted.cosmetic.<id> permissions,
# the same ones Tebex sets when a cosmetic is bought on its own.
lp creategroup vip
lp group vip parent add default
lp group vip meta setsuffix 20 " <#99ff33>ᴠɪᴘ</#99ff33>"
lp group vip permission set mostwanted.cosmetic.escape-confetti true

lp creategroup vip_plus
lp group vip_plus parent add vip
lp group vip_plus meta setsuffix 21 " <#33ffcc>ᴠɪᴘ+</#33ffcc>"
lp group vip_plus permission set mostwanted.cosmetic.trail-smoke true
lp group vip_plus permission set mostwanted.cosmetic.chat-colours true

lp creategroup mvp
lp group mvp parent add vip_plus
lp group mvp meta setsuffix 22 " <#6633ff>ᴍᴠᴘ</#6633ff>"
lp group mvp permission set mostwanted.cosmetic.trail-cash true
lp group mvp permission set mostwanted.cosmetic.trail-ember true
lp group mvp permission set mostwanted.cosmetic.trail-siren true
lp group mvp permission set mostwanted.cosmetic.escape-fireworks true
lp group mvp permission set mostwanted.cosmetic.escape-thunder true
lp group mvp permission set mostwanted.cosmetic.arrest-spotlight true
lp group mvp permission set mostwanted.cosmetic.arrest-siren true
lp group mvp permission set mostwanted.name-colour true
lp group mvp permission set mostwanted.join-announce true

lp creategroup legend
lp group legend parent add mvp
lp group legend meta setsuffix 23 " <#ff3366>ʟᴇɢᴇɴᴅ</#ff3366>"
lp group legend permission set mostwanted.cosmetic.* true
lp group legend permission set mostwanted.join-message true
